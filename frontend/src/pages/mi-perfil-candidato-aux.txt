import React from 'react';
import { ChevronLeftIcon } from 'lucide-react';
import { ProfileHeader } from '../components/ProfileHeader';
import { SkillsList } from '../components/SkillsList';
import { ExperienceSection } from '../components/ExperienceSection';
import { ContactSidebar } from '../components/ContactSidebar';
import { useCandidato } from '../hooks/useCandidato';

// TODO: Reemplazar con el context real de autenticación.
const CURRENT_USER_ID = 'user-1';

interface MiPerfilCandidatoProps {
  /** ID del candidato a mostrar. Por defecto muestra el usuario actual. */
  candidatoId?: string;
  onVolver?: () => void;
  onEditar?: () => void;
  onDescargarCV?: () => void;
  /** TODO: Navegar a /candidato/:id/experiencia/:expId */
  onVerMasExperiencia?: (expId: string) => void;
}

export const MiPerfilCandidato: React.FC<MiPerfilCandidatoProps> = ({
  candidatoId = CURRENT_USER_ID,
  onVolver,
  onEditar,
  onDescargarCV,
  onVerMasExperiencia,
}) => {
  // TODO: Reemplazar con llamada a API real: GET /api/candidatos/{id}
  const { candidato, loading, error } = useCandidato(candidatoId);

  // El propietario del perfil puede editar y descargar CV
  const esDueño = candidatoId === CURRENT_USER_ID;

  return (
    <div className="min-h-screen flex flex-col bg-[#EFEFEF]">

      {/* ── Navbar institucional ── */}
      <header className="w-full bg-[#1E2749] px-6 md:px-10 py-4 flex items-center gap-4">
        <div className="flex flex-col gap-[5px] cursor-pointer">
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
        </div>
        <div className="flex flex-col leading-none ml-1">
          <span className="font-bold text-white text-xl tracking-tight leading-tight">
            Portal de Empleos
          </span>
          <span className="font-normal text-[#a8b4d4] text-sm tracking-wide leading-tight">
            Instituto Madero
          </span>
        </div>
      </header>

      {/* ── Hero / título ── */}
      <section className="w-full bg-[#1E2749] px-6 py-10 pb-12 flex flex-col items-center">
        <h1 className="font-bold text-white text-3xl md:text-[34px] tracking-tight text-center">
          Mi perfil
        </h1>
      </section>

      {/* ── Franja "Volver al inicio" ── */}
      <div className="w-full bg-white px-6 md:px-10 py-3 border-b border-[#eeeeee]">
        <button
          onClick={onVolver}
          className="flex items-center gap-1 text-[#3351A6] text-sm font-medium hover:opacity-75 transition-opacity"
        >
          <ChevronLeftIcon className="w-4 h-4" />
          Volver al inicio
        </button>
      </div>

      {/* ── Contenido principal ── */}
      <main className="flex-1 px-4 md:px-8 lg:px-12 py-8">
        {loading ? (
          <p className="text-center text-[#888888] text-sm py-24">Cargando perfil...</p>
        ) : error ? (
          <p className="text-center text-[#F46036] text-sm py-24">{error}</p>
        ) : candidato ? (
          <div className="max-w-[1000px] mx-auto flex flex-col lg:flex-row gap-5 items-start">

            {/* ── Columna izquierda ── */}
            <div className="flex flex-col gap-5 flex-1 w-full min-w-0">

              {/* Card: nombre + título */}
              <ProfileHeader
                nombre={candidato.nombre}
                titulo={candidato.titulo}
                esDueño={esDueño}
                onEditar={onEditar}
              />

              {/* Card: bio + habilidades */}
              <div className="bg-white rounded-xl shadow-sm border border-[#e2e2e2] px-6 py-6 flex flex-col gap-5">
                <div className="flex flex-col gap-2">
                  <h2 className="font-bold text-[#1a1a2e] text-[15px]">
                    Información personal
                  </h2>
                  <p className="text-[#666666] text-sm leading-relaxed">{candidato.bio}</p>
                </div>
                <div className="flex flex-col gap-3">
                  <h2 className="font-bold text-[#1a1a2e] text-[15px]">Habilidades</h2>
                  <SkillsList habilidades={candidato.habilidades} />
                </div>
              </div>

              {/* Card: experiencia + educación */}
              <ExperienceSection
                experiencia={candidato.experiencia}
                educacion={candidato.educacion}
                onVerMasExperiencia={onVerMasExperiencia}
              />
            </div>

            {/* ── Columna derecha ── */}
            <div className="w-full lg:w-[280px] flex-shrink-0">
              <ContactSidebar
                ubicacion={candidato.ubicacion}
                edad={candidato.edad}
                contacto={candidato.contacto}
                esDueño={esDueño}
                onDescargarCV={onDescargarCV}
              />
            </div>

          </div>
        ) : null}
      </main>

      {/* ── Footer institucional ── */}
      <footer className="w-full bg-[#1E2749] px-4 py-6 text-center mt-auto">
        <p className="text-white text-sm font-medium">
          &copy; 2025 Portal de Empleos del Instituto Madero.
        </p>
        <p className="text-[#6e7f9e] text-xs mt-1">
          Desarrollado por estudiantes de la Tecnicatura Universitaria en Programación &mdash; UTN FRBA.
        </p>
      </footer>
    </div>
  );
};
