import React, { useState } from 'react';
import { ChevronLeftIcon, ChevronDownIcon } from 'lucide-react';

type TipoTrabajo = '' | 'full_time' | 'part_time' | 'freelance' | 'pasantia';

interface CrearOfertaForm {
  tituloPuesto: string;
  descripcion: string;
  ubicacion: string;
  tipoTrabajo: TipoTrabajo;
  sueldoIndicado: string;
}

const TIPOS_TRABAJO: { value: TipoTrabajo; label: string }[] = [
  { value: 'full_time', label: 'Full-time' },
  { value: 'part_time', label: 'Part-time' },
  { value: 'freelance', label: 'Freelance' },
  { value: 'pasantia', label: 'Pasantía' },
];

const INITIAL_FORM: CrearOfertaForm = {
  tituloPuesto: '',
  descripcion: '',
  ubicacion: '',
  tipoTrabajo: '',
  sueldoIndicado: '',
};

interface CrearOfertaProps {
  onVolver?: () => void;
  // TODO: Reemplazar con llamada a API real: POST /api/reclutador/ofertas
  onPublicar?: (data: CrearOfertaForm) => Promise<void>;
}

export const CrearOferta: React.FC<CrearOfertaProps> = ({ onVolver, onPublicar }) => {
  const [form, setForm] = useState<CrearOfertaForm>(INITIAL_FORM);
  const [submitting, setSubmitting] = useState(false);
  const [submitError, setSubmitError] = useState<string | null>(null);
  const [submitted, setSubmitted] = useState(false);

  const handleChange = (
    e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>
  ) => {
    const { name, value } = e.target;
    setForm((prev) => ({ ...prev, [name]: value }));
    if (submitError) setSubmitError(null);
  };

  const isValid =
    form.tituloPuesto.trim() !== '' &&
    form.descripcion.trim() !== '' &&
    form.ubicacion.trim() !== '' &&
    form.tipoTrabajo !== '' &&
    form.sueldoIndicado.trim() !== '';

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!isValid || submitting) return;
    setSubmitting(true);
    setSubmitError(null);
    try {
      if (onPublicar) {
        await onPublicar(form);
      }
      setSubmitted(true);
      setForm(INITIAL_FORM);
    } catch {
      setSubmitError('Ocurrió un error al publicar la oferta. Intentá de nuevo.');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <div className="min-h-screen flex flex-col bg-[#EFEFEF]">

      {/* ── Navbar institucional ── */}
      <header className="w-full bg-[#1E2749] px-6 md:px-10 py-4 flex items-center gap-4">
        <div className="flex flex-col gap-[5px] cursor-pointer">
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
        </div>
        <div className="flex flex-col leading-none ml-1">
          <span className="font-bold text-white text-xl tracking-tight leading-tight">
            Portal de Empleos
          </span>
          <span className="font-normal text-[#a8b4d4] text-sm tracking-wide leading-tight">
            Instituto Madero
          </span>
        </div>
      </header>

      {/* ── Hero / título ── */}
      <section className="w-full bg-[#1E2749] px-6 py-10 flex flex-col items-center gap-2 pb-12">
        <h1 className="font-bold text-white text-3xl md:text-[34px] tracking-tight text-center">
          Crear oferta laboral
        </h1>
        <p className="text-[#a8b4d4] text-base md:text-[17px] text-center">
          Completá el formulario para publicar una nueva búsqueda laboral
        </p>
      </section>

      {/* ── Contenido principal ── */}
      <main className="flex-1 px-4 md:px-8 lg:px-16 py-6">

        {/* Volver al inicio */}
        <button
          onClick={onVolver}
          className="flex items-center gap-1 text-[#3351A6] text-sm font-medium hover:opacity-75 transition-opacity mb-8"
        >
          <ChevronLeftIcon className="w-4 h-4" />
          Volver al inicio
        </button>

        {/* ── Card del formulario ── */}
        <div className="bg-white rounded-xl shadow-sm border border-[#e2e2e2] max-w-[540px] mx-auto px-8 py-8">

          {submitted ? (
            <div className="flex flex-col items-center gap-3 py-8">
              <div className="w-14 h-14 rounded-full bg-green-100 flex items-center justify-center">
                <svg className="w-7 h-7 text-green-600" viewBox="0 0 24 24" fill="none">
                  <path d="M5 13l4 4L19 7" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
                </svg>
              </div>
              <p className="font-bold text-[#1E2749] text-lg text-center">¡Oferta publicada!</p>
              <p className="text-[#666666] text-sm text-center">La oferta ya es visible para los candidatos.</p>
              <button
                onClick={() => setSubmitted(false)}
                className="mt-2 text-[#3351A6] text-sm font-semibold hover:opacity-75 transition-opacity"
              >
                Crear otra oferta
              </button>
            </div>
          ) : (
            <form onSubmit={handleSubmit} noValidate className="flex flex-col gap-5">

              {/* Título del puesto */}
              <div className="flex flex-col gap-1.5">
                <label className="text-[#333333] text-sm font-medium">
                  Título del puesto{' '}
                  <span className="text-[#E53E3E]">*</span>
                </label>
                <input
                  type="text"
                  name="tituloPuesto"
                  value={form.tituloPuesto}
                  onChange={handleChange}
                  placeholder="Ej: Desarrollador Web Senior"
                  maxLength={100}
                  className="w-full border border-[#d0d0d0] rounded-lg px-4 py-2.5 text-sm text-[#333333] placeholder:text-[#aaaaaa] outline-none focus:border-[#3351A6] focus:ring-1 focus:ring-[#3351A6]/30 transition-colors"
                />
              </div>

              {/* Descripción */}
              <div className="flex flex-col gap-1.5">
                <label className="text-[#333333] text-sm font-medium">
                  Descripción{' '}
                  <span className="text-[#E53E3E]">*</span>
                </label>
                <textarea
                  name="descripcion"
                  value={form.descripcion}
                  onChange={handleChange}
                  placeholder="Describí las tareas, requisitos y beneficios del puesto..."
                  rows={4}
                  maxLength={2000}
                  className="w-full border border-[#d0d0d0] rounded-lg px-4 py-2.5 text-sm text-[#333333] placeholder:text-[#aaaaaa] outline-none focus:border-[#3351A6] focus:ring-1 focus:ring-[#3351A6]/30 transition-colors resize-y min-h-[100px]"
                />
              </div>

              {/* Ubicación + Tipo de trabajo (2 columnas) */}
              <div className="flex flex-col sm:flex-row gap-4">
                <div className="flex flex-col gap-1.5 flex-1">
                  <label className="text-[#333333] text-sm font-medium">
                    Ubicación{' '}
                    <span className="text-[#E53E3E]">*</span>
                  </label>
                  <input
                    type="text"
                    name="ubicacion"
                    value={form.ubicacion}
                    onChange={handleChange}
                    placeholder="Ej: Buenos Aires"
                    maxLength={100}
                    className="w-full border border-[#d0d0d0] rounded-lg px-4 py-2.5 text-sm text-[#333333] placeholder:text-[#aaaaaa] outline-none focus:border-[#3351A6] focus:ring-1 focus:ring-[#3351A6]/30 transition-colors"
                  />
                </div>

                <div className="flex flex-col gap-1.5 flex-1">
                  <label className="text-[#333333] text-sm font-medium">
                    Tipo de trabajo{' '}
                    <span className="text-[#E53E3E]">*</span>
                  </label>
                  <div className="relative">
                    <select
                      name="tipoTrabajo"
                      value={form.tipoTrabajo}
                      onChange={handleChange}
                      className="w-full appearance-none border border-[#d0d0d0] rounded-lg px-4 py-2.5 text-sm text-[#333333] outline-none focus:border-[#3351A6] focus:ring-1 focus:ring-[#3351A6]/30 transition-colors bg-white cursor-pointer pr-9"
                    >
                      <option value="" disabled>Seleccioná el tipo</option>
                      {TIPOS_TRABAJO.map((t) => (
                        <option key={t.value} value={t.value}>
                          {t.label}
                        </option>
                      ))}
                    </select>
                    <ChevronDownIcon className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[#888888] pointer-events-none" />
                  </div>
                </div>
              </div>

              {/* Sueldo indicado */}
              <div className="flex flex-col gap-1.5">
                <label className="text-[#333333] text-sm font-medium">
                  Sueldo indicado{' '}
                  <span className="text-[#E53E3E]">*</span>
                </label>
                <input
                  type="text"
                  name="sueldoIndicado"
                  value={form.sueldoIndicado}
                  onChange={handleChange}
                  placeholder="Ej: $120.000"
                  maxLength={30}
                  className="w-full border border-[#d0d0d0] rounded-lg px-4 py-2.5 text-sm text-[#333333] placeholder:text-[#aaaaaa] outline-none focus:border-[#3351A6] focus:ring-1 focus:ring-[#3351A6]/30 transition-colors"
                />
              </div>

              {/* Nota informativa */}
              <div className="bg-[#EBF2FB] border border-[#C8D9F0] rounded-lg px-4 py-3">
                <p className="text-[#3351A6] text-xs leading-relaxed">
                  Los campos marcados con <span className="font-semibold">*</span> son obligatorios.
                  La oferta será visible de inmediato para los candidatos.
                </p>
              </div>

              {/* Error de envío */}
              {submitError && (
                <p className="text-[#E53E3E] text-xs text-center font-medium">{submitError}</p>
              )}

              {/* Botón publicar */}
              <button
                type="submit"
                disabled={!isValid || submitting}
                className={`w-full py-3.5 rounded-lg font-bold text-sm tracking-widest text-white uppercase transition-all duration-200 ${
                  !isValid || submitting
                    ? 'bg-[#F46036]/50 cursor-not-allowed'
                    : 'bg-[#F46036] hover:bg-[#e2552f] active:scale-[0.99] cursor-pointer'
                }`}
              >
                {submitting ? 'Publicando...' : 'Publicar oferta'}
              </button>

            </form>
          )}
        </div>
      </main>

      {/* ── Footer institucional ── */}
      <footer className="w-full bg-[#1E2749] px-4 py-6 text-center mt-auto">
        <p className="text-white text-sm font-medium">
          &copy; 2025 Portal de Empleos del Instituto Madero.
        </p>
        <p className="text-[#6e7f9e] text-xs mt-1">
          Desarrollado por estudiantes de la Tecnicatura Universitaria en Programación &mdash; UTN FRBA.
        </p>
      </footer>
    </div>
  );
};
