import React, { useState, useMemo } from 'react';
import {
  ChevronLeftIcon,
  ChevronRightIcon,
  MapPinIcon,
  CalendarIcon,
  BuildingIcon,
} from 'lucide-react';
import { EstadoBadge } from '../components/EstadoBadge';
import { usePostulaciones } from '../hooks/usePostulaciones';
import type { FiltroEstado } from '../hooks/usePostulaciones';

const ITEMS_PER_PAGE = 6;

interface FiltroTab {
  label: string;
  value: FiltroEstado;
  classes: string;
}

const FILTROS: FiltroTab[] = [
  {
    label: 'Mis postulaciones',
    value: 'todos',
    classes: 'bg-white text-[#555555] border border-[#d0d0d0]',
  },
  {
    label: 'Pendiente',
    value: 'pendiente',
    classes: 'bg-[#FFF9E6] text-[#7A6000] border border-[#F5E0A3]',
  },
  {
    label: 'En revisión',
    value: 'en_revision',
    classes: 'bg-[#D8E6F7] text-[#1A2F5E] border border-[#B5CDEB]',
  },
  {
    label: 'Rechazado',
    value: 'rechazado',
    classes: 'bg-[#FCE8E6] text-[#A8201A] border border-[#F7C4C0]',
  },
];

const formatFecha = (iso: string): string => {
  const [year, month, day] = iso.split('-');
  return `${day}/${month}/${year}`;
};

interface MisPostulacionesProps {
  onVolver?: () => void;
  onVerDetalle?: (id: string) => void;
}

export const MisPostulaciones: React.FC<MisPostulacionesProps> = ({
  onVolver,
  onVerDetalle,
}) => {
  const [filtroEstado, setFiltroEstado] = useState<FiltroEstado>('todos');
  const [currentPage, setCurrentPage] = useState(1);

  const { postulaciones, loading, error } = usePostulaciones(filtroEstado);

  const totalPages = Math.ceil(postulaciones.length / ITEMS_PER_PAGE);
  const startIndex = (currentPage - 1) * ITEMS_PER_PAGE;
  const paginatedItems = useMemo(
    () => postulaciones.slice(startIndex, startIndex + ITEMS_PER_PAGE),
    [postulaciones, startIndex]
  );

  const handleFiltroChange = (value: FiltroEstado) => {
    setFiltroEstado(value);
    setCurrentPage(1);
  };

  const handlePageChange = (page: number) => {
    if (page < 1 || page > totalPages) return;
    setCurrentPage(page);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  return (
    <div className="min-h-screen flex flex-col bg-[#EFEFEF]">

      {/* ── Navbar institucional ── */}
      <header className="w-full bg-[#1E2749] px-6 md:px-10 py-4 flex items-center gap-4">
        <div className="flex flex-col gap-[5px] cursor-pointer">
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
        </div>
        <div className="flex flex-col leading-none ml-1">
          <span className="font-bold text-white text-xl tracking-tight leading-tight">
            Portal de Empleos
          </span>
          <span className="font-normal text-[#a8b4d4] text-sm tracking-wide leading-tight">
            Instituto Madero
          </span>
        </div>
      </header>

      {/* ── Hero / título ── */}
      <section className="w-full bg-[#1E2749] px-6 py-10 flex flex-col items-center gap-2 pb-12">
        <h1 className="font-bold text-white text-3xl md:text-[34px] tracking-tight text-center">
          Mis postulaciones
        </h1>
        <p className="text-[#a8b4d4] text-base md:text-[17px] text-center">
          Seguí el estado de cada empleo al que te postulaste
        </p>
      </section>

      {/* ── Contenido principal ── */}
      <main className="flex-1 px-4 md:px-8 lg:px-16 py-6">

        {/* Volver al inicio */}
        <button
          onClick={onVolver}
          className="flex items-center gap-1 text-[#3351A6] text-sm font-medium hover:opacity-75 transition-opacity mb-5"
        >
          <ChevronLeftIcon className="w-4 h-4" />
          Volver al inicio
        </button>

        {/* ── Filtros (tabs con color del estado) ── */}
        <div className="flex flex-wrap items-center gap-2 mb-5">
          {FILTROS.map((f) => (
            <button
              key={f.value}
              onClick={() => handleFiltroChange(f.value)}
              className={`px-5 py-2 rounded-full text-sm font-medium transition-all duration-150 cursor-pointer outline-none
                ${f.classes}
                ${filtroEstado === f.value ? 'ring-2 ring-offset-1 ring-[#3351A6]/40 font-semibold' : 'hover:opacity-80'}
              `}
            >
              {f.label}
            </button>
          ))}
        </div>

        {/* ── Tabla ── */}
        <div className="bg-white rounded-xl shadow-sm border border-[#e2e2e2] overflow-hidden">

          {loading ? (
            <div className="flex items-center justify-center py-24">
              <p className="text-[#888888] text-sm">Cargando postulaciones...</p>
            </div>
          ) : error ? (
            <div className="flex items-center justify-center py-24">
              <p className="text-[#F46036] text-sm">{error}</p>
            </div>
          ) : postulaciones.length === 0 ? (
            <div className="flex flex-col items-center justify-center py-24 gap-2">
              <p className="text-[#999999] text-sm font-medium">
                No hay postulaciones para este filtro
              </p>
            </div>
          ) : (
            <>
              {/* Desktop table */}
              <div className="hidden md:block overflow-x-auto">
                <table className="w-full">
                  <thead>
                    <tr className="border-b border-[#eeeeee]">
                      <th className="px-6 py-4 text-left text-[11px] font-bold text-[#888888] tracking-widest uppercase">
                        Puesto
                      </th>
                      <th className="px-6 py-4 text-left text-[11px] font-bold text-[#888888] tracking-widest uppercase">
                        Empresa
                      </th>
                      <th className="px-6 py-4 text-left text-[11px] font-bold text-[#888888] tracking-widest uppercase">
                        Ubicación
                      </th>
                      <th className="px-6 py-4 text-left text-[11px] font-bold text-[#888888] tracking-widest uppercase">
                        Fecha
                      </th>
                      <th className="px-6 py-4 text-left text-[11px] font-bold text-[#888888] tracking-widest uppercase">
                        Estado
                      </th>
                      <th className="px-6 py-4" />
                    </tr>
                  </thead>
                  <tbody>
                    {paginatedItems.map((postulacion, idx) => (
                      <tr
                        key={postulacion.id}
                        className={`${idx < paginatedItems.length - 1 ? 'border-b border-[#f0f0f0]' : ''} hover:bg-[#fafafa] transition-colors`}
                      >
                        {/* Puesto */}
                        <td className="px-6 py-[18px]">
                          <span className="font-bold text-[#1a1a2e] text-[14px] leading-snug">
                            {postulacion.puesto}
                          </span>
                        </td>

                        {/* Empresa */}
                        <td className="px-6 py-[18px]">
                          <div className="flex items-center gap-2">
                            <BuildingIcon className="w-4 h-4 text-[#888888] flex-shrink-0" />
                            <span className="text-[#444444] text-[14px]">{postulacion.empresa}</span>
                          </div>
                        </td>

                        {/* Ubicación */}
                        <td className="px-6 py-[18px]">
                          <div className="flex items-center gap-2">
                            <MapPinIcon className="w-4 h-4 text-[#888888] flex-shrink-0" />
                            <span className="text-[#444444] text-[14px]">{postulacion.ubicacion.split(',')[0]}</span>
                          </div>
                        </td>

                        {/* Fecha */}
                        <td className="px-6 py-[18px]">
                          <div className="flex items-center gap-2">
                            <CalendarIcon className="w-4 h-4 text-[#888888] flex-shrink-0" />
                            <span className="text-[#444444] text-[14px] whitespace-nowrap">
                              {formatFecha(postulacion.fecha)}
                            </span>
                          </div>
                        </td>

                        {/* Estado */}
                        <td className="px-6 py-[18px]">
                          <EstadoBadge estado={postulacion.estado} />
                        </td>

                        {/* Ver más */}
                        <td className="px-6 py-[18px]">
                          <button
                            onClick={() => onVerDetalle?.(postulacion.id)}
                            className="flex items-center gap-1 text-[#3351A6] text-[14px] font-semibold hover:opacity-70 transition-opacity cursor-pointer whitespace-nowrap"
                          >
                            Ver más
                            <ChevronRightIcon className="w-4 h-4" />
                          </button>
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>

              {/* Mobile cards */}
              <div className="md:hidden divide-y divide-[#f0f0f0]">
                {paginatedItems.map((postulacion) => (
                  <div key={postulacion.id} className="px-4 py-4 flex flex-col gap-2">
                    <div className="flex items-start justify-between gap-2">
                      <span className="font-bold text-[#1a1a2e] text-sm leading-snug flex-1">
                        {postulacion.puesto}
                      </span>
                      <EstadoBadge estado={postulacion.estado} />
                    </div>
                    <div className="flex items-center gap-1.5 text-[#666666] text-xs">
                      <BuildingIcon className="w-3.5 h-3.5 flex-shrink-0" />
                      <span>{postulacion.empresa}</span>
                    </div>
                    <div className="flex items-center gap-1.5 text-[#888888] text-xs">
                      <MapPinIcon className="w-3.5 h-3.5 flex-shrink-0" />
                      <span>{postulacion.ubicacion}</span>
                    </div>
                    <div className="flex items-center gap-1.5 text-[#aaaaaa] text-xs">
                      <CalendarIcon className="w-3.5 h-3.5 flex-shrink-0" />
                      <span>{formatFecha(postulacion.fecha)}</span>
                    </div>
                    <button
                      onClick={() => onVerDetalle?.(postulacion.id)}
                      className="self-start flex items-center gap-1 text-[#3351A6] text-sm font-semibold hover:opacity-70 transition-opacity mt-1"
                    >
                      Ver más <ChevronRightIcon className="w-3.5 h-3.5" />
                    </button>
                  </div>
                ))}
              </div>

              {/* ── Paginación ── */}
              {totalPages > 1 && (
                <div className="flex items-center justify-center gap-1 py-5 border-t border-[#f0f0f0]">
                  <button
                    onClick={() => handlePageChange(currentPage - 1)}
                    disabled={currentPage === 1}
                    className={`w-9 h-9 flex items-center justify-center rounded transition-colors ${
                      currentPage === 1
                        ? 'text-[#cccccc] cursor-not-allowed'
                        : 'text-[#F46036] hover:bg-[#fff5f2] cursor-pointer'
                    }`}
                    aria-label="Página anterior"
                  >
                    <ChevronLeftIcon className="w-4 h-4" />
                  </button>

                  {Array.from({ length: totalPages }, (_, i) => i + 1).map((page) => (
                    <button
                      key={page}
                      onClick={() => handlePageChange(page)}
                      className={`w-9 h-9 flex items-center justify-center rounded text-sm font-semibold transition-colors cursor-pointer ${
                        currentPage === page
                          ? 'bg-[#F46036] text-white rounded-full'
                          : 'text-[#444444] hover:bg-[#f5f5f5]'
                      }`}
                      aria-label={`Página ${page}`}
                      aria-current={currentPage === page ? 'page' : undefined}
                    >
                      {page}
                    </button>
                  ))}

                  <button
                    onClick={() => handlePageChange(currentPage + 1)}
                    disabled={currentPage === totalPages}
                    className={`w-9 h-9 flex items-center justify-center rounded transition-colors ${
                      currentPage === totalPages
                        ? 'text-[#cccccc] cursor-not-allowed'
                        : 'text-[#F46036] hover:bg-[#fff5f2] cursor-pointer'
                    }`}
                    aria-label="Página siguiente"
                  >
                    <ChevronRightIcon className="w-4 h-4" />
                  </button>
                </div>
              )}
            </>
          )}
        </div>
      </main>

      {/* ── Footer institucional ── */}
      <footer>
        <p className="text-white text-sm font-medium">
          &copy; 2025 Portal de Empleos del Instituto Madero.
        </p>
        <p className="text-[#6e7f9e] text-xs mt-1">
          Desarrollado por estudiantes de la Tecnicatura Universitaria en Programación &mdash; UTN FRBA.
        </p>
      </footer>
    </div>
  );
};
