// TODO: Proteger esta ruta con el middleware de autenticación de roles.
// Esta pantalla es exclusiva del rol 'reclutador'.

import React, { useState, useCallback } from 'react';
import { ChevronLeftIcon } from 'lucide-react';
import { CandidatoCard } from '../components/CandidatoCard';
import { mockCandidatosRecibidos } from '../data/mockData';
import type { CandidatoRecibido, EstadoCandidato } from '../types/postulacion';

// TODO: Reemplazar con el hook/context real de autenticación.
const USER_ROLE: 'reclutador' | 'candidato' = 'reclutador';

// Datos de la oferta asociada a estas postulaciones.
// TODO: Reemplazar con lookup dinámico por ofertaId desde la URL/contexto.
const OFERTA_TITULO = 'Desarrollador Web';
const OFERTA_EMPRESA = 'TechCorp';

interface PostulacionesRecibidasProps {
  onVolver?: () => void;
  onVerPerfil?: (candidatoId: string) => void;
}

export const PostulacionesRecibidas: React.FC<PostulacionesRecibidasProps> = ({
  onVolver,
  onVerPerfil,
}) => {
  // Estado local de los candidatos — permite cambio de estado visual inmediato.
  // TODO: Reemplazar con llamada a API real: GET /api/ofertas/{ofertaId}/postulaciones
  const [candidatos, setCandidatos] = useState<CandidatoRecibido[]>(
    mockCandidatosRecibidos
  );

  const updateEstado = useCallback(
    (id: string, nuevoEstado: EstadoCandidato) => {
      setCandidatos((prev) =>
        prev.map((c) => (c.id === id ? { ...c, estado: nuevoEstado } : c))
      );
    },
    []
  );

  const handleAccept = useCallback(
    (id: string) => {
      // TODO: Reemplazar con llamada a API real: PUT /api/postulaciones/{id}/estado
      updateEstado(id, 'aceptado');
    },
    [updateEstado]
  );

  const handleReject = useCallback(
    (id: string) => {
      // TODO: Reemplazar con llamada a API real: PUT /api/postulaciones/{id}/estado
      updateEstado(id, 'rechazado');
    },
    [updateEstado]
  );

  const handleRevisar = useCallback(
    (id: string) => {
      // TODO: Reemplazar con llamada a API real: PUT /api/postulaciones/{id}/estado
      updateEstado(id, 'en_revision');
    },
    [updateEstado]
  );

  const handleVerPerfil = useCallback(
    (id: string) => {
      // TODO: Navegar a /candidato/:id con react-router cuando esté integrado.
      if (onVerPerfil) {
        onVerPerfil(id);
      }
    },
    [onVerPerfil]
  );

  // Protección de rol en el cliente
  if (USER_ROLE !== 'reclutador') {
    return (
      <div className="min-h-screen flex items-center justify-center bg-[#EFEFEF]">
        <p className="text-[#F46036] font-semibold text-base">
          Acceso restringido: esta pantalla es solo para reclutadores.
        </p>
      </div>
    );
  }

  return (
    <div className="min-h-screen flex flex-col bg-[#EFEFEF]">

      {/* ── Navbar institucional ── */}
      <header className="w-full bg-[#1E2749] px-6 md:px-10 py-4 flex items-center gap-4">
        <div className="flex flex-col gap-[5px] cursor-pointer">
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
          <span className="block w-[22px] h-[2px] bg-white rounded-sm" />
        </div>
        <div className="flex flex-col leading-none ml-1">
          <span className="font-bold text-white text-xl tracking-tight leading-tight">
            Portal de Empleos
          </span>
          <span className="font-normal text-[#a8b4d4] text-sm tracking-wide leading-tight">
            Instituto Madero
          </span>
        </div>
      </header>

      {/* ── Hero / título ── */}
      <section className="w-full bg-[#1E2749] px-6 py-10 flex flex-col items-center gap-2 pb-12">
        <h1 className="font-bold text-white text-3xl md:text-[34px] tracking-tight text-center">
          Postulaciones recibidas
        </h1>
        <p className="text-[#a8b4d4] text-base text-center">
          {OFERTA_TITULO} – {OFERTA_EMPRESA}
          <span className="mx-2 opacity-50">·</span>
          {candidatos.length} candidato{candidatos.length !== 1 ? 's' : ''}
        </p>
      </section>

      {/* ── Franja "Volver al inicio" (fondo blanco) ── */}
      <div className="w-full bg-white px-6 md:px-10 py-3 border-b border-[#eeeeee]">
        <button
          onClick={onVolver}
          className="flex items-center gap-1 text-[#3351A6] text-sm font-medium hover:opacity-75 transition-opacity"
        >
          <ChevronLeftIcon className="w-4 h-4" />
          Volver al inicio
        </button>
      </div>

      {/* ── Listado de candidatos ── */}
      <main className="flex-1 px-4 md:px-8 lg:px-16 py-8">
        {candidatos.length === 0 ? (
          <p className="text-center text-[#aaaaaa] text-sm py-16">
            No hay postulaciones recibidas para esta oferta.
          </p>
        ) : (
          <div className="flex flex-col gap-5 max-w-[660px] mx-auto">
            {candidatos.map((candidato) => (
              <CandidatoCard
                key={candidato.id}
                candidato={candidato}
                onVerPerfil={handleVerPerfil}
                onAccept={handleAccept}
                onReject={handleReject}
                onRevisar={handleRevisar}
              />
            ))}
          </div>
        )}
      </main>

      {/* ── Footer institucional ── */}
      <footer className="w-full bg-[#1E2749] px-4 py-6 text-center mt-auto">
        <p className="text-white text-sm font-medium">
          &copy; 2025 Portal de Empleos del Instituto Madero.
        </p>
        <p className="text-[#6e7f9e] text-xs mt-1">
          Desarrollado por estudiantes de la Tecnicatura Universitaria en Programación &mdash; UTN FRBA.
        </p>
      </footer>
    </div>
  );
};
